DECLARE @UserId int = 42;

;WITH AccessDetails AS (
SELECT
u.Id AS UserId,
u.UserName,
u.IsGlobalAdministrator,
ur.RoleId,
b.Name AS BranchName,
ur.DepartmentId,
d.Name AS DepartmentName,
r.Name AS RoleName,
ap.Code AS PolicyCode,
p.Name AS PolicyPermission,
CASE WHEN rp.PermissionId IS NOT NULL THEN 1 ELSE 0 END AS GrantedByRole,
apr.MessageTypePattern,
apr.Direction,
apr.RequiresReference
FROM orp.Users AS u
LEFT JOIN orp.UserRoles AS ur ON ur.UserId = u.Id
LEFT JOIN orp.Branches AS b ON b.Id = ur.BranchId
LEFT JOIN orp.Departments AS d ON d.Id = ur.DepartmentId
LEFT JOIN orp.Roles AS r ON r.Id = ur.RoleId
LEFT JOIN orp.AccessPolicies AS ap ON ap.Id = ur.AccessPolicyId
LEFT JOIN orp.AccessPolicyRules AS apr ON apr.AccessPolicyId = ap.Id
LEFT JOIN orp.Permissions AS p ON p.Id = apr.PermissionId
LEFT JOIN orp.RolePermissions AS rp
ON rp.RoleId = ur.RoleId AND rp.PermissionId = apr.PermissionId
WHERE u.Id = @UserId
),
AccessSummary AS (
SELECT
COUNT(DISTINCT DepartmentId) AS DepartmentCount,
MAX(GrantedByRole) AS HasEffectivePermission,
MAX(CASE WHEN GrantedByRole = 1 AND PolicyPermission IN (
N'review.level1', N'review.level2', N'review.level3', N'review.bulk-approve'
) THEN 1 ELSE 0 END) AS HasReviewPermission
FROM AccessDetails
)
SELECT
ad.*,
s.DepartmentCount,
CASE
WHEN ad.IsGlobalAdministrator = 1 THEN N'GLOBAL_ADMIN'
WHEN ad.RoleId IS NULL THEN N'NO_ROLE_ASSIGNMENTS'
WHEN s.DepartmentCount > 1 AND s.HasReviewPermission = 1
THEN N'REVIEWER_IN_MULTIPLE_DEPARTMENTS'
WHEN s.HasEffectivePermission = 0 THEN N'NO_EFFECTIVE_PERMISSIONS'
ELSE N'HAS_EFFECTIVE_PERMISSIONS'
END AS Diagnosis
FROM AccessDetails AS ad
CROSS JOIN AccessSummary AS s
ORDER BY ad.BranchName, ad.DepartmentName, ad.RoleName,
ad.PolicyCode, ad.PolicyPermission;


WITH DepartmentAccess AS (
SELECT
ur.UserId,
ur.DepartmentId,
MAX(CASE WHEN rp.PermissionId IS NOT NULL THEN 1 ELSE 0 END)
AS HasEffectivePermission,
MAX(CASE WHEN p.Name IN (
N'review.level1', N'review.level2', N'review.level3', N'review.bulk-approve'
) THEN 1 ELSE 0 END) AS HasReviewPermission
FROM orp.UserRoles AS ur
LEFT JOIN orp.AccessPolicyRules AS apr
ON apr.AccessPolicyId = ur.AccessPolicyId
LEFT JOIN orp.RolePermissions AS rp
ON rp.RoleId = ur.RoleId AND rp.PermissionId = apr.PermissionId
LEFT JOIN orp.Permissions AS p ON p.Id = rp.PermissionId
GROUP BY ur.UserId, ur.DepartmentId
)
SELECT
u.Id AS UserId,
u.UserName,
u.DisplayName,
COUNT(*) AS AssignedDepartments,
SUM(da.HasEffectivePermission) AS DepartmentsWithPermissions,
STRING_AGG(CONCAT(da.DepartmentId, N' — ', d.Name), N', ')
WITHIN GROUP (ORDER BY da.DepartmentId) AS Departments,
CAST(CASE WHEN u.IsGlobalAdministrator = 0
AND MAX(da.HasReviewPermission) = 1
THEN 1 ELSE 0 END AS bit) AS BlockedReviewer
FROM DepartmentAccess AS da
JOIN orp.Users AS u ON u.Id = da.UserId
JOIN orp.Departments AS d ON d.Id = da.DepartmentId
GROUP BY u.Id, u.UserName, u.DisplayName, u.IsGlobalAdministrator
HAVING COUNT(*) > 1
ORDER BY BlockedReviewer DESC, AssignedDepartments DESC, u.Id;
