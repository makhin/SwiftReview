DECLARE @UserId int = NULL;
DECLARE @DepartmentName nvarchar(80) = N'CS';

DECLARE @DepartmentId int;
DECLARE @PolicyId int;
DECLARE @IsGlobalAdministrator bit;
DECLARE @Roles TABLE (Id int NULL);

BEGIN TRY
BEGIN TRANSACTION;

    SELECT @IsGlobalAdministrator = IsGlobalAdministrator
    FROM orp.Users WITH (UPDLOCK, HOLDLOCK)
    WHERE Id = @UserId;

    IF @IsGlobalAdministrator IS NULL
        THROW 51000, 'Set @UserId to an existing orp.Users.Id.', 1;

    IF @IsGlobalAdministrator = 1
        THROW 51000, 'Choose an account that uses scoped roles. This account is a Global Administrator.', 1;

    SELECT @DepartmentId = MIN(Id)
    FROM orp.Departments WITH (HOLDLOCK)
    WHERE Name = TRIM(@DepartmentName)
    HAVING COUNT(*) = 1;

    IF @DepartmentId IS NULL
        THROW 51000, '@DepartmentName must identify exactly one department.', 1;

    SELECT @PolicyId = Id FROM orp.AccessPolicies WITH (HOLDLOCK) WHERE Code = N'FULL';
    IF @PolicyId IS NULL
        THROW 51000, 'The FULL access policy is missing.', 1;

    INSERT INTO @Roles (Id)
    SELECT DISTINCT r.Id
    FROM STRING_SPLIT(@RoleNames, N',') s
    LEFT JOIN orp.Roles r WITH (HOLDLOCK) ON r.Name = TRIM(s.value)
    WHERE TRIM(s.value) <> N'';

    IF NOT EXISTS (SELECT 1 FROM @Roles) OR EXISTS (SELECT 1 FROM @Roles WHERE Id IS NULL)
        THROW 51000, '@RoleNames must contain one or more role names from orp.Roles.', 1;

    IF EXISTS
    (
        SELECT rp.PermissionId
        FROM orp.RolePermissions rp WITH (HOLDLOCK)
        JOIN @Roles r ON r.Id = rp.RoleId
        EXCEPT
        SELECT PermissionId
        FROM orp.AccessPolicyRules WITH (HOLDLOCK)
        WHERE AccessPolicyId = @PolicyId AND MessageTypePattern = N'%'
            AND Direction = 0 AND RequiresReference = 0
    )
        THROW 51000, 'The FULL policy does not cover all permissions of the requested roles.', 1;

    IF NOT EXISTS (SELECT 1 FROM orp.Branches WITH (HOLDLOCK))
        THROW 51000, 'No branches found.', 1;

    -- Keep only existing assignments that already match the requested department, policy and roles.
    DELETE ur
    FROM orp.UserRoles ur WITH (UPDLOCK, HOLDLOCK)
    WHERE ur.UserId = @UserId
        AND (ur.DepartmentId <> @DepartmentId OR ur.AccessPolicyId <> @PolicyId
            OR NOT EXISTS (SELECT 1 FROM @Roles r WHERE r.Id = ur.RoleId));

    INSERT INTO orp.UserRoles (UserId, BranchId, DepartmentId, RoleId, AccessPolicyId)
    SELECT @UserId, b.Id, @DepartmentId, r.Id, @PolicyId
    FROM orp.Branches b WITH (HOLDLOCK)
    CROSS JOIN @Roles r
    WHERE NOT EXISTS
    (
        SELECT 1 FROM orp.UserRoles ur WITH (UPDLOCK, HOLDLOCK)
        WHERE ur.UserId = @UserId AND ur.BranchId = b.Id AND ur.DepartmentId = @DepartmentId
            AND ur.RoleId = r.Id AND ur.AccessPolicyId = @PolicyId
    );

    -- Apply the same active-review check as the ORP access administration API.
    IF EXISTS
    (
        SELECT 1
        FROM orp.Reviews rv WITH (HOLDLOCK)
        JOIN orp.Messages m WITH (HOLDLOCK) ON m.MessageId = rv.MessageId
        WHERE rv.ReviewerId = @UserId AND rv.Status = N'InProgress'
            AND NOT EXISTS
            (
                SELECT 1 FROM orp.MessageDepartments d WITH (HOLDLOCK)
                WHERE d.MessageId = m.MessageId AND d.CanProcess = 1
                    AND (rv.DepartmentId IS NULL OR d.DepartmentId = rv.DepartmentId)
                    AND (m.ReviewDepartmentId IS NULL OR d.DepartmentId = m.ReviewDepartmentId)
                    AND orp.RoleAllows(@UserId, m.MessageId, d.DepartmentId,
                        CASE rv.Level WHEN 1 THEN N'review.level1' WHEN 2 THEN N'review.level2' ELSE N'review.level3' END) = 1
            )
    )
        THROW 51000, 'These changes would prevent the user from completing an active review.', 1;

    COMMIT TRANSACTION;
END TRY

THROW;
